struct EffectRequest {
    tool: string,
    risk: string,
    user_authorized: bool
}

fn admit(request: EffectRequest)
    -> Result<Option<string>, string> {
    let mut decision: Result<Option<string>, string> =
        Result::Ok(None);
    if request.tool == "" {
        decision = Result::Err("missing capability");
    }
    if request.tool != "" && request.risk == "high"
        && !request.user_authorized {
        decision = Result::Ok(Some("needs_user"));
    }
    decision
}
